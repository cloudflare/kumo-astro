## Mobile Dividers

<p>
  In Astro, <code>variant</code> and <code>mobileDivider</code> must be passed directly to each
  <code>GridItem</code> — there is no React context to propagate them automatically.
</p>

<ComponentExample
  code={`<Grid variant="4up">\n  <GridItem variant="4up" mobileDivider>Item 1</GridItem>\n  <GridItem variant="4up" mobileDivider>Item 2</GridItem>\n  <GridItem variant="4up" mobileDivider>Item 3</GridItem>\n  <GridItem variant="4up" mobileDivider>Item 4</GridItem>\n</Grid>`}
  lang="astro"
>
  <Grid variant="4up">
    <GridItem variant="4up" mobileDivider><div class="rounded-lg bg-kumo-tint p-4 text-sm">Item 1</div></GridItem>
    <GridItem variant="4up" mobileDivider><div class="rounded-lg bg-kumo-tint p-4 text-sm">Item 2</div></GridItem>
    <GridItem variant="4up" mobileDivider><div class="rounded-lg bg-kumo-tint p-4 text-sm">Item 3</div></GridItem>
    <GridItem variant="4up" mobileDivider><div class="rounded-lg bg-kumo-tint p-4 text-sm">Item 4</div></GridItem>
  </Grid>
</ComponentExample>
