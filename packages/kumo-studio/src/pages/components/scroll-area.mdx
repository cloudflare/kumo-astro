## Examples

### Vertical

<ComponentExample
  code={`<ScrollArea maxHeight={200} class="w-48 rounded-md border border-kumo-line p-3">\n  <!-- 20 items -->\n</ScrollArea>`}
  lang="astro"
>
  <ScrollArea maxHeight={200} class="w-48 rounded-md border border-kumo-line p-3">
    <div class="flex flex-col gap-1">
      {Array.from({ length: 20 }, (_, i) => (
        <p class="py-0.5 text-sm text-kumo-default">v1.{i}.0</p>
      ))}
    </div>
  </ScrollArea>
</ComponentExample>

### Horizontal

<ComponentExample
  code={`<ScrollArea orientation="horizontal" maxWidth="100%" class="rounded-md border border-kumo-line p-3">\n  <div class="flex gap-4" style="width:max-content">\n    <!-- wide content -->\n  </div>\n</ScrollArea>`}
  lang="astro"
>
  <ScrollArea orientation="horizontal" maxWidth="100%" class="rounded-md border border-kumo-line p-3">
    <div class="flex gap-4" style="width:max-content">
      {Array.from({ length: 15 }, (_, i) => (
        <div class="shrink-0 rounded-md border border-kumo-line px-3 py-1 text-sm text-kumo-default">
          v1.{i}.0
        </div>
      ))}
    </div>
  </ScrollArea>
</ComponentExample>

### Both Directions

<ComponentExample
  code={`<ScrollArea orientation="both" maxHeight={200} maxWidth="100%" class="rounded-md border border-kumo-line p-3">\n  <!-- wide and tall content -->\n</ScrollArea>`}
  lang="astro"
>
  <ScrollArea orientation="both" maxHeight={200} maxWidth="100%" class="rounded-md border border-kumo-line p-3">
    <div style="width:600px">
      {Array.from({ length: 15 }, (_, i) => (
        <p class="py-1 text-sm text-kumo-default whitespace-nowrap">
          Row {i + 1} — This is a long line of content that extends beyond the visible area
        </p>
      ))}
    </div>
  </ScrollArea>
</ComponentExample>
