## Installation

### Barrel

```tsx
import { Resizable } from "@cloudflare/kumo";
```

### Granular

```tsx
import { Resizable } from "@cloudflare/kumo/components/resizable";
```

## Usage

```tsx
import { Resizable } from "@cloudflare/kumo";

export default function Example() {
  return (
    <Resizable className="h-40">
      <Resizable.Panel defaultSize={50}>Left</Resizable.Panel>
      <Resizable.Handle />
      <Resizable.Panel defaultSize={50}>Right</Resizable.Panel>
    </Resizable>
  );
}
```
