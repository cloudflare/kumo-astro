## Installation

### Barrel

```tsx
import { HoverCard } from "@cloudflare/kumo";
```

### Granular

```tsx
import { HoverCard } from "@cloudflare/kumo/components/hover-card";
```

## Usage

```tsx
import { HoverCard } from "@cloudflare/kumo";

export default function Example() {
  return (
    <div className="relative inline-block">
      <HoverCard>
        <HoverCard.Trigger>
          <a href="#">@username</a>
        </HoverCard.Trigger>
        <HoverCard.Content>
          <p>User profile preview</p>
        </HoverCard.Content>
      </HoverCard>
    </div>
  );
}
```
