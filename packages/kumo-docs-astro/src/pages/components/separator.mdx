## Installation

### Barrel

```tsx
import { Separator } from "@cloudflare/kumo";
```

### Granular

```tsx
import { Separator } from "@cloudflare/kumo/components/separator";
```

## Usage

```tsx
import { Separator } from "@cloudflare/kumo";

export default function Example() {
  return (
    <div>
      <p>Above</p>
      <Separator />
      <p>Below</p>
    </div>
  );
}
```
