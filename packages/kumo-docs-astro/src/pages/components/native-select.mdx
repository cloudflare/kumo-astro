## Installation

### Barrel

```tsx
import { NativeSelect } from "@cloudflare/kumo";
```

### Granular

```tsx
import { NativeSelect } from "@cloudflare/kumo/components/native-select";
```

## Usage

```tsx
import { NativeSelect } from "@cloudflare/kumo";

export default function Example() {
  return (
    <NativeSelect aria-label="Region">
      <option value="us">United States</option>
      <option value="eu">Europe</option>
    </NativeSelect>
  );
}
```
