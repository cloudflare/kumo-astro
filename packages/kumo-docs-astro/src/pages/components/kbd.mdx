## Installation

### Barrel

```tsx
import { Kbd } from "@cloudflare/kumo";
```

### Granular

```tsx
import { Kbd } from "@cloudflare/kumo/components/kbd";
```

## Usage

```tsx
import { Kbd } from "@cloudflare/kumo";

export default function Example() {
  return (
    <p>
      Press <Kbd>⌘K</Kbd> to search
    </p>
  );
}
```
