## Installation

### Barrel

```tsx
import { NavigationMenu } from "@cloudflare/kumo";
```

### Granular

```tsx
import { NavigationMenu } from "@cloudflare/kumo/components/navigation-menu";
```

## Usage

```tsx
import { NavigationMenu } from "@cloudflare/kumo";

export default function Example() {
  return (
    <NavigationMenu>
      <NavigationMenu.List>
        <NavigationMenu.Item>
          <NavigationMenu.Link href="/" active>Home</NavigationMenu.Link>
        </NavigationMenu.Item>
        <NavigationMenu.Item value="products">
          <NavigationMenu.Trigger>Products</NavigationMenu.Trigger>
          <NavigationMenu.Content>
            <NavigationMenu.SubLink href="/workers" title="Workers" description="Edge functions" />
          </NavigationMenu.Content>
        </NavigationMenu.Item>
      </NavigationMenu.List>
    </NavigationMenu>
  );
}
```
