## Installation

### Barrel

```tsx
import { ScrollArea } from "@cloudflare/kumo";
```

### Granular

```tsx
import { ScrollArea } from "@cloudflare/kumo/components/scroll-area";
```

## Usage

```tsx
import { ScrollArea } from "@cloudflare/kumo";

export default function Example() {
  return (
    <ScrollArea maxHeight={300}>
      <ul>
        {items.map(item => (
          <li key={item.id}>{item.name}</li>
        ))}
      </ul>
    </ScrollArea>
  );
}
```
