## Installation

### Barrel

```tsx
import { AspectRatio } from "@cloudflare/kumo";
```

### Granular

```tsx
import { AspectRatio } from "@cloudflare/kumo/components/aspect-ratio";
```

## Usage

```tsx
import { AspectRatio } from "@cloudflare/kumo";

export default function Example() {
  return (
    <AspectRatio ratio="16/9">
      <img src="..." className="h-full w-full object-cover rounded-lg" />
    </AspectRatio>
  );
}
```
